== top-level exports object preserves original key order ==
{
  "version": "1.0.0",
  "exports": {
    "types": "./dist/index.d.ts",
    "default": "./dist/index.js",
    "./feature": "./dist/feature.js",
    ".": "./dist/index.mjs",
    "import": "./dist/index.js"
  },
  "name": "pkg"
}

[expect]
{
  "name": "pkg",
  "version": "1.0.0",
  "exports": {
    "types": "./dist/index.d.ts",
    "default": "./dist/index.js",
    "./feature": "./dist/feature.js",
    ".": "./dist/index.mjs",
    "import": "./dist/index.js"
  }
}
